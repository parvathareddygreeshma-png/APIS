def get_db():
    conn = sqlite3.connect(DATABASE)
    conn.row_factory = sqlite3.Row
    return conn


def initialize_database():

    conn = get_db()
    cursor = conn.cursor()

    cursor.execute("""
        CREATE TABLE IF NOT EXISTS access_points (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT,
            location TEXT,
            band TEXT,
            channel INTEGER,
            users INTEGER,
            signal REAL,
            throughput REAL,
            latency REAL,
            packet_loss REAL,
            status TEXT
        )
    """)

    cursor.execute("""
        CREATE TABLE IF NOT EXISTS configuration (
            id INTEGER PRIMARY KEY,
            ssid TEXT,
            channel_width TEXT,
            transmit_power INTEGER,
            ofdma TEXT,
            mu_mimo TEXT,
            twt TEXT
        )
    """)

    cursor.execute("SELECT COUNT(*) FROM access_points")

    if cursor.fetchone()[0] == 0:

        access_points = [
            ("AP-01", "Main Block", "5 GHz", 36, 42, -48, 820, 18, 0.8, "Online"),
            ("AP-02", "Computer Lab", "5 GHz", 44, 58, -52, 760, 24, 1.2, "Online"),
            ("AP-03", "Library", "5 GHz", 149, 35, -45, 910, 15, 0.5, "Online"),
            ("AP-04", "Seminar Hall", "5 GHz", 157, 74, -67, 540, 48, 3.8, "Warning"),
            ("AP-05", "Hostel Block", "2.4 GHz", 6, 63, -61, 310, 55, 4.2, "Warning"),
            ("AP-06", "Cafeteria", "5 GHz", 40, 29, -50, 780, 20, 1.0, "Online")
        ]

        cursor.executemany("""
            INSERT INTO access_points
            (name, location, band, channel, users, signal,
             throughput, latency, packet_loss, status)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
        """, access_points)

    cursor.execute("SELECT COUNT(*) FROM configuration")

    if cursor.fetchone()[0] == 0:

        cursor.execute("""
            INSERT INTO configuration
            VALUES (1, 'SmartCampus-WiFi6', '80 MHz',
                    20, 'Enabled', 'Enabled', 'Enabled')
        """)

    conn.commit()
    conn.close()


initialize_database()

# ============================================================
# ACCESS POINT API

@app.route("/api/access-points")
def access_points():

    conn = get_db()

    rows = conn.execute(
        "SELECT * FROM access_points"
    ).fetchall()

    conn.close()

    return jsonify([
        dict(row)
        for row in rows
    ])


# ============================================================
# OPTIMIZATION API
# ============================================================

@app.route("/api/optimize", methods=["POST"])
def optimize():

    conn = get_db()

    aps = conn.execute(
        "SELECT * FROM access_points"
    ).fetchall()

    recommendations = []

    for ap in aps:

        if ap["users"] > 65:

            recommendations.append(
                f"{ap['name']}: Reduce congestion using AP load balancing."
            )

        if ap["latency"] > 40:

            recommendations.append(
                f"{ap['name']}: High latency detected. "
                "Recommend channel optimization."
            )

        if ap["signal"] < -60:

            recommendations.append(
                f"{ap['name']}: Weak signal detected. "
                "Increase AP coverage or adjust transmit power."
            )

        if ap["packet_loss"] > 3:

            recommendations.append(
                f"{ap['name']}: Packet loss is high. "
                "Investigate interference."
            )


    if not recommendations:

        recommendations.append(
            "Network performance is within recommended limits."
        )


    conn.close()

    return jsonify({
        "recommendations":
        recommendations
    })


# ============================================================

@app.route(
    "/api/configuration",
    methods=["POST"]
)
def configuration():

    data = request.get_json()

    conn = get_db()

    conn.execute("""
        UPDATE configuration
        SET ssid = ?,
            channel_width = ?,
            transmit_power = ?,
            ofdma = ?,
            mu_mimo = ?,
            twt = ?
        WHERE id = 1
    """, (

        data["ssid"],
        data["width"],
        data["power"],
        data["ofdma"],
        data["mimo"],
        data["twt"]

    ))

    conn.commit()
    conn.close()

    return jsonify({
        "success": True
    })
